def _regenerate_step_with_feedback(self, step_id, validation_result, test_case_id):
    """Regenerate a step based on validation feedback."""
    feedback_prompt = f"""
The previously generated step had validation issues:
Errors: {', '.join(validation_result.errors)}
Warnings: {', '.join(validation_result.warnings)}

Please regenerate this step addressing these issues:
{chr(10).join(f'- {s}' for s in validation_result.suggestions)}
"""
    
    # Call AI with feedback
    new_step = self.ai_helper.generate_step_with_feedback(
        test_case_id=test_case_id,
        feedback=feedback_prompt
    )
    
    return new_step

def execute_step_with_retry(self, step, env_helper, browser, max_retries=3):
    """Execute step with retry on failure."""
    for attempt in range(1, max_retries + 1):
        try:
            return self.execute_step(step, env_helper, browser)
        except Exception as e:
            if attempt == max_retries:
                raise
            
            self.logger.warning(f"Step failed (attempt {attempt}/{max_retries}), retrying...")
            
            # Get error category
            error_msg = str(e)
            error_category = self.feedback_collector.categorize_error(error_msg)
            
            # Apply recovery strategy based on error
            if error_category == 'selector_not_found':
                # Try CSS selector fallback if available
                if step.get('css_selector'):
                    step['element_locator'] = step['css_selector']
                    step['by_strategy'] = 'css'
                    continue
            
            elif error_category == 'stale_element':
                # Add wait before retry
                import time
                time.sleep(1)
                continue
            
            elif error_category == 'element_not_clickable':
                # Try scrolling to element
                self._scroll_to_element(step['element_locator'])
                continue
            
            # Default: just retry
            continue
```

# In main.py or monitoring service
@app.get("/api/agent/metrics")
async def get_agent_metrics(current_user: User = Depends(get_current_user)):
    """Get Phase 1 agent metrics."""
    with get_db_connection_context() as conn:
        with conn.cursor() as cursor:
            # Validation metrics
            cursor.execute("""
                SELECT 
                    COUNT(*) as total_validations,
                    SUM(CASE WHEN is_valid THEN 1 ELSE 0 END) as valid_steps,
                    AVG(confidence) as avg_confidence
                FROM validation_results
                WHERE test_case_id IN (
                    SELECT id FROM test_cases WHERE client_id = %s
                )
            """, (current_user.client_id,))
            
            validation_metrics = cursor.fetchone()
            
            # Feedback metrics
            cursor.execute("""
                SELECT 
                    COUNT(*) as total_failures,
                    COUNT(DISTINCT error_type) as error_categories
                FROM execution_feedback
                WHERE test_case_id IN (
                    SELECT id FROM test_cases WHERE client_id = %s
                )
            """, (current_user.client_id,))
            
            feedback_metrics = cursor.fetchone()
            
            return {
                'validation': {
                    'total': validation_metrics[0],
                    'valid': validation_metrics[1],
                    'avg_confidence': validation_metrics[2]
                },
                'feedback': {
                    'total_failures': feedback_metrics[0],
                    'error_categories': feedback_metrics[1]
                }
            }
```
